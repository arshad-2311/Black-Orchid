    68→    68→export function Sparkline({ data, color = "#d4af37" }: { data: number[]; color?: string }) {
    69→    69→  const max = Math.max(...data, 1);
    70→    70→  const min = Math.min(...data, 0);
    71→    71→  const range = max - min || 1;
    72→    72→  const pts = data.map((v, i) => {
    73→    73→    const x = (i / (data.length - 1)) * 100;
    74→    74→    const y = 100 - ((v - min) / range) * 100;
    75→    75→    return `${x},${y}`;
    76→    76→  }).join(" ");
    77→    77→  return (
    78→    78→    <svg viewBox="0 0 100 100" preserveAspectRatio="none" className="h-full w-full">
    79→    79→      <defs>
    80→    80→        <linearGradient id={`spark-${color.replace("#", "")}`} x1="0" y1="0" x2="0" y2="1">
    81→    81→          <stop offset="0%" stopColor={color} stopOpacity="0.35" />
    82→    82→          <stop offset="100%" stopColor={color} stopOpacity="0" />
    83→    83→        </linearGradient>
    84→    84→      </defs>
    85→    85→      <polygon points={`0,100 ${pts} 100,100`} fill={`url(#spark-${color.replace("#", "")})`} />
    86→    86→      <polyline points={pts} fill="none" stroke={color} strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" vectorEffect="non-scaling-stroke" />
    87→    87→    </svg>
    88→    88→  );
    89→    89→}

   222→   222→type Option = { value: string; label: string };

   224→   224→export function SearchableSelect({
   225→   225→  label, required, options, value, onChange, placeholder = "Select…", error, searchable = true,
   226→   226→}: {
   227→   227→  label?: string; required?: boolean; options: Option[]; value: string; onChange: (v: string) => void;
   228→   228→  placeholder?: string; error?: string; searchable?: boolean;
   229→   229→}) {
   230→   230→  const [open, setOpen] = useState(false);
   231→   231→  const [query, setQuery] = useState("");
   232→   232→  const [active, setActive] = useState(0);
   233→   233→  const ref = useRef<HTMLDivElement | null>(null);
   234→   234→
   235→   235→  const selected = options.find((o) => o.value === value);
   236→   236→  const filtered = options.filter((o) => o.label.toLowerCase().includes(query.toLowerCase()));
   237→   237→
   238→   238→  useEffect(() => {
   239→   239→    const onDoc = (e: MouseEvent) => { if (ref.current && !ref.current.contains(e.target as Node)) setOpen(false); };
   240→   240→    document.addEventListener("mousedown", onDoc);
   241→   241→    return () => document.removeEventListener("mousedown", onDoc);
   242→   242→  }, []);
   243→   243→
   244→   244→  const choose = (v: string) => { onChange(v); setOpen(false); setQuery(""); };
   245→   245→
   246→   246→  const onKey = (e: React.KeyboardEvent) => {
   247→   247→    if (!open && (e.key === "ArrowDown" || e.key === "Enter")) { setOpen(true); return; }
   248→   248→    if (e.key === "ArrowDown") { e.preventDefault(); setActive((a) => Math.min(a + 1, filtered.length - 1)); }
   249→   249→    if (e.key === "ArrowUp") { e.preventDefault(); setActive((a) => Math.max(a - 1, 0)); }
   250→   250→    if (e.key === "Enter" && open) { e.preventDefault(); if (filtered[active]) choose(filtered[active].value); }
   251→   251→    if (e.key === "Escape") setOpen(false);
   252→   252→  };
   253→   253→
   254→   254→  return (
   255→   255→    <div className="block">
   256→   256→      {label && <span className="mb-1.5 flex items-center gap-1 admin-label">{label}{required && <span className="text-admin-gold">*</span>}</span>}
   257→   257→      <div ref={ref} className="relative" onKeyDown={onKey}>
   258→   258→        <button
   259→   259→          type="button"
   260→   260→          onClick={() => setOpen((o) => !o)}
   261→   261→          className={cn("admin-input flex items-center justify-between text-left", !selected && "text-admin-muted", error && "border-red-500/60")}
   262→   262→        >
   263→   263→          <span className="truncate">{selected ? selected.label : placeholder}</span>
   264→   264→          <ChevronDown className={cn("h-4 w-4 shrink-0 text-admin-muted transition-transform", open && "rotate-180")} />
   265→   265→        </button>
   266→   266→        <AnimatePresence>
   267→   267→          {open && (
   268→   268→            <motion.div
   269→   269→              initial={{ opacity: 0, y: -6, scale: 0.98 }}
   270→   270→              animate={{ opacity: 1, y: 0, scale: 1 }}
   271→   271→              exit={{ opacity: 0, y: -6, scale: 0.98 }}
   272→   272→              transition={{ duration: 160, ease: [0.22, 1, 0.36, 1] }}
   273→   273→              className="admin-surface-elevated absolute z-50 mt-2 w-full overflow-hidden p-1.5"
   274→   274→            >
   275→   275→              {searchable && (
   276→   276→                <div className="relative mb-1.5">
   277→   277→                  <Search className="pointer-events-none absolute left-3 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-admin-muted" />
   278→   278→                  <input
   279→   279→                    autoFocus
   280→   280→                    value={query}
   281→   281→                    onChange={(e) => { setQuery(e.target.value); setActive(0); }}
   282→   282→                    placeholder="Search…"
   283→   283→                    className="h-9 w-full rounded-lg border border-admin-border bg-admin-bg/60 pl-9 pr-3 text-sm text-admin-text outline-none focus:border-admin-gold/50"
   284→   284→                  />
   285→   285→                </div>
   286→   286→              )}
   287→   287→              <div className="max-h-56 overflow-y-auto">
   288→   288→                {filtered.length === 0 ? (
   289→   289→                  <p className="px-3 py-6 text-center font-sans text-sm text-admin-muted">No results</p>
   290→   290→                ) : filtered.map((o, i) => (
   291→   291→                  <button
   292→   292→                    key={o.value}
   293→   293→                    type="button"
   294→   294→                    onMouseEnter={() => setActive(i)}
   295→   295→                    onClick={() => choose(o.value)}
   296→   296→                    className={cn(
   297→   297→                      "flex w-full items-center justify-between rounded-lg px-3 py-2 text-left font-sans text-sm transition-colors",
   298→   298→                      i === active ? "bg-admin-gold/15 text-admin-gold" : "text-admin-text hover:bg-white/5"
   299→   299→                    )}
   300→   300→                  >
   301→   301→                    {o.label}
   302→   302→                    {o.value === value && <Check className="h-4 w-4" />}
   303→   303→                  </button>
   304→   304→                ))}
   305→   305→              </div>
   306→   306→            </motion.div>
   307→   307→          )}
   308→   308→        </AnimatePresence>
   309→   309→      </div>
   310→   310→      {error && <p className="mt-1.5 flex items-center gap-1 font-sans text-xs text-red-400"><AlertTriangle className="h-3 w-3" />{error}</p>}
   311→   311→    </div>
   312→   312→  );
   313→   313→}

   345→   345→export function AdminButton({
   346→   346→  children, variant = "solid", className, size = "md", confirm, onConfirm, ...props
   347→   347→}: {
   348→   348→  variant?: "solid" | "outline" | "danger" | "ghost" | "subtle"; size?: "sm" | "md" | "lg";
   349→   349→  confirm?: string; onConfirm?: () => void;

   351→   351→  const sizes = {
   352→   352→    sm: "h-8 px-3 text-[11px]",
   353→   353→    md: "h-10 px-4 text-xs",
   354→   354→    lg: "h-12 px-6 text-sm",
   355→   355→  } as const;

   359→   359→    danger: "border border-red-500/30 bg-red-500/10 text-red-400 hover:bg-red-500/20 hover:border-red-500/50",

   367→   367→      return;
   368→   368→    }
   369→   369→    props.onClick?.(e);
   370→   370→  };
   371→   371→
   372→   372→  return (
   373→   373→    <button
   374→   374→      {...props}
   375→   375→      onClick={handleClick}
   376→   376→      className={cn(
   377→   377→        "inline-flex items-center justify-center gap-1.5 rounded-xl font-sans uppercase tracking-wider transition-all duration-200 disabled:opacity-50 disabled:pointer-events-none",
   378→   378→        sizes[size], styles[variant], className
   379→   379→      )}
   380→   380→    >
   381→   381→      {children}
   382→   382→    </button>

   422→   422→export function ImageUploader({
   423→   423→  value, onChange, label = "Image", aspect = "16/10",
   424→   424→}: {
   425→   425→  value: string; onChange: (v: string) => void; label?: string; aspect?: string;
   426→   426→}) {
   427→   427→  const [dragging, setDragging] = useState(false);
   428→   428→  const [progress, setProgress] = useState<number | null>(null);
   429→   429→  const [error, setError] = useState("");
   430→   430→  const [urlMode, setUrlMode] = useState(false);
   431→   431→  const [urlInput, setUrlInput] = useState("");
   432→   432→  const inputRef = useRef<HTMLInputElement | null>(null);
   433→   433→
   434→   434→  const validate = (file: File) => {
   435→   435→    const okType = /image\/(jpeg|jpg|png|webp|gif|avif)/.test(file.type);
   436→   436→    const okSize = file.size <= 6 * 1024 * 1024;
   437→   437→    if (!okType) { setError("Only JPG, PNG, WebP, GIF, AVIF allowed"); return false; }
   438→   438→    if (!okSize) { setError("Max file size is 6MB"); return false; }
   439→   439→    setError(""); return true;
   440→   440→  };
   441→   441→
   442→   442→  const handleFile = useCallback((file: File) => {
   443→   443→    if (!validate(file)) return;
   444→   444→    const reader = new FileReader();
   445→   445→    reader.onprogress = (e) => { if (e.lengthComputable) setProgress(Math.round((e.loaded / e.total) * 100)); };
   446→   446→    reader.onload = () => {
   447→   447→      setProgress(100);
   448→   448→      setTimeout(() => {
   449→   449→        setProgress(null);
   450→   450→        onChange(reader.result as string);
   451→   451→      }, 300);
   452→   452→    };
   453→   453→    reader.readAsDataURL(file);
   454→   454→  }, [onChange]);
   455→   455→
   456→   456→  const onDrop = (e: React.DragEvent) => {
   457→   457→    e.preventDefault(); setDragging(false);
   458→   458→    const file = e.dataTransfer.files?.[0];
   459→   459→    if (file) handleFile(file);
   460→   460→  };
   461→   461→
   462→   462→  const hasImage = !!value;
   463→   463→
   464→   464→  return (
   465→   465→    <div className="block">
   466→   466→      <div className="mb-1.5 flex items-center justify-between">
   467→   467→        <span className="admin-label">{label}</span>
   468→   468→        <button type="button" onClick={() => setUrlMode((m) => !m)} className="font-sans text-[11px] text-admin-gold hover:underline">
   469→   469→          {urlMode ? "Upload file" : "Paste URL"}
   470→   470→        </button>
   471→   471→      </div>
   472→   472→
   473→   473→      {urlMode ? (
   474→   474→        <div className="flex gap-2">
   475→   475→          <input
   476→   476→            value={urlInput}
   477→   477→            onChange={(e) => setUrlInput(e.target.value)}
   478→   478→            placeholder="https://example.com/image.jpg"
   479→   479→            className="admin-input flex-1"
   480→   480→          />
   481→   481→          <AdminButton type="button" variant="subtle" onClick={() => { if (urlInput) { onChange(urlInput); setUrlInput(""); } }}>Set</AdminButton>
   482→   482→        </div>
   483→   483→      ) : !hasImage ? (
   484→   484→        <div
   485→   485→          onDragOver={(e) => { e.preventDefault(); setDragging(true); }}
   486→   486→          onDragLeave={() => setDragging(false)}
   487→   487→          onDrop={onDrop}
   488→   488→          onClick={() => inputRef.current?.click()}
   489→   489→          className={cn(
   490→   490→            "relative flex cursor-pointer flex-col items-center justify-center rounded-xl border-2 border-dashed px-6 py-10 text-center transition-all",
   491→   491→            dragging ? "border-admin-gold bg-admin-gold/10" : "border-admin-border bg-admin-bg/40 hover:border-admin-gold/40 hover:bg-admin-gold/5"
   492→   492→          )}
   493→   493→          style={{ aspectRatio: aspect }}
   494→   494→        >
   495→   495→          <input ref={inputRef} type="file" accept="image/*" className="hidden" onChange={(e) => { const f = e.target.files?.[0]; if (f) handleFile(f); e.target.value = ""; }} />
   496→   496→          <div className="flex h-12 w-12 items-center justify-center rounded-full border border-admin-border bg-admin-gold/10 text-admin-gold">
   497→   497→            <Upload className="h-5 w-5" />
   498→   498→          </div>
   499→   499→          <p className="mt-3 font-sans text-sm font-medium text-admin-text">Drop image here, or <span className="text-admin-gold">browse</span></p>
   500→   500→          <p className="mt-1 font-sans text-xs text-admin-muted">JPG, PNG, WebP, GIF · up to 6MB</p>
   501→   501→        </div>
   502→   502→      ) : (
   503→   503→        <div className="group relative overflow-hidden rounded-xl border border-admin-border" style={{ aspectRatio: aspect }}>
   504→   504→          <img src={value} alt="Preview" className="h-full w-full object-cover" />
   505→   505→          <div className="absolute inset-0 flex items-center justify-center gap-2 bg-black/60 opacity-0 backdrop-blur-sm transition-opacity duration-200 group-hover:opacity-100">
   506→   506→            <AdminButton type="button" size="sm" variant="subtle" onClick={() => inputRef.current?.click()}><ImageIcon className="h-3.5 w-3.5" /> Replace</AdminButton>
   507→   507→            <AdminButton type="button" size="sm" variant="danger" onClick={() => onChange("")}><Trash2 className="h-3.5 w-3.5" /> Remove</AdminButton>
   508→   508→          </div>
   509→   509→          <input ref={inputRef} type="file" accept="image/*" className="hidden" onChange={(e) => { const f = e.target.files?.[0]; if (f) handleFile(f); e.target.value = ""; }} />
   510→   510→        </div>
   511→   511→      )}
   512→   512→
   513→   513→      {/* Progress bar */}
   514→   514→      {progress !== null && (
   515→   515→        <div className="mt-2">
   516→   516→          <div className="h-1.5 w-full overflow-hidden rounded-full bg-white/10">
   517→   517→            <motion.div className="admin-gold-bg h-full" initial={{ width: 0 }} animate={{ width: `${progress}%` }} transition={{ duration: 0.2 }} />
   518→   518→          </div>
   519→   519→          <p className="mt-1 text-center font-sans text-[11px] text-admin-muted">Uploading… {progress}%</p>
   520→   520→        </div>
   521→   521→      )}
   522→   522→      {error && <p className="mt-1.5 flex items-center gap-1 font-sans text-xs text-red-400"><AlertTriangle className="h-3 w-3" />{error}</p>}
   523→   523→    </div>
   524→   524→  );
   525→   525→}

   553→   553→export function Pagination({ page, totalPages, onPage }: { page: number; totalPages: number; onPage: (p: number) => void }) {
   554→   554→  if (totalPages <= 1) return null;
   555→   555→  return (
   556→   556→    <div className="flex items-center justify-between gap-3 px-1 py-3">
   557→   557→      <p className="font-sans text-xs text-admin-muted">Page {page} of {totalPages}</p>
   558→   558→      <div className="flex gap-1.5">
   559→   559→        <button onClick={() => onPage(page - 1)} disabled={page <= 1} className="flex h-8 w-8 items-center justify-center rounded-lg border border-admin-border text-admin-muted transition-colors hover:bg-white/5 disabled:opacity-30 disabled:hover:bg-transparent">
   560→   560→          <ChevronLeft className="h-4 w-4" />
   561→   561→        </button>
   562→   562→        {Array.from({ length: totalPages }).map((_, i) => (
   563→   563→          <button
   564→   564→            key={i}
   565→   565→            onClick={() => onPage(i + 1)}
   566→   566→            className={cn("h-8 w-8 rounded-lg border font-sans text-xs transition-colors", page === i + 1 ? "admin-gold-bg border-transparent font-semibold text-black" : "border-admin-border text-admin-muted hover:bg-white/5")}
   567→   567→          >
   568→   568→            {i + 1}
   569→   569→          </button>
   570→   570→        ))}
   571→   571→        <button onClick={() => onPage(page + 1)} disabled={page >= totalPages} className="flex h-8 w-8 items-center justify-center rounded-lg border border-admin-border text-admin-muted transition-colors hover:bg-white/5 disabled:opacity-30 disabled:hover:bg-transparent">
   572→   572→          <ChevronRight className="h-4 w-4" />
   573→   573→        </button>
   574→   574→      </div>
   575→   575→    </div>
   576→   576→  );
   577→   577→}
