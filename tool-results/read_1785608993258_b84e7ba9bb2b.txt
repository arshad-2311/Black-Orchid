    44→
    45→export function ReservationView() {
    46→  const [step, setStep] = useState(0);
    47→  const [direction, setDirection] = useState(1);
    48→  const [form, setForm] = useState<FormState>(EMPTY_FORM);
    49→  const [errors, setErrors] = useState<Record<string, string>>({});
    50→  const [loading, setLoading] = useState(false);
    51→  const [success, setSuccess] = useState<Reservation | null>(null);
    52→  const wizardRef = useRef<HTMLDivElement>(null);
    53→
    54→  const today = new Date().toISOString().slice(0, 10);
    55→
    56→  const update = (key: keyof FormState, value: string) => {
    57→    setForm((f) => ({ ...f, [key]: value }));
    58→    setErrors((e) => (e[key] ? { ...e, [key]: "" } : e));
    59→  };
    60→
    61→  const validateStep = (s: number): Record<string, string> => {
    62→    const e: Record<string, string> = {};
    63→    if (s === 0) {
    64→      // Date
    65→      if (!form.date) e.date = "Please select a date";
    66→    } else if (s === 1) {
    67→      // Time
    68→      if (!form.time) e.time = "Please select a time";
    69→    } else if (s === 2) {
    70→      // Guests — always valid (defaults to 2), no validation needed
    71→    } else if (s === 3) {
    72→      // Customer details
    73→      if (!form.name.trim()) e.name = "Please enter your name";
    74→      else if (form.name.trim().length < 2) e.name = "Name is too short";
    75→      if (!form.phone.trim()) e.phone = "Please enter your phone number";
    76→      else if (form.phone.replace(/\D/g, "").length < 7) e.phone = "Enter a valid phone number";
    77→      if (!form.email.trim()) e.email = "Please enter your email";
    78→      else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email.trim())) e.email = "Enter a valid email address";
    79→    }
    80→    return e;
    81→  };
    82→
    83→  const scrollToWizard = () => {
    84→    if (wizardRef.current && typeof window !== "undefined") {
    85→      const top = wizardRef.current.getBoundingClientRect().top + window.scrollY - 90;
    86→      window.scrollTo({ top, behavior: "smooth" });
    87→    }
    88→  };
    89→
    90→  const next = () => {
    91→    const e = validateStep(step);
    92→    setErrors(e);
    93→    if (Object.keys(e).length > 0) return;
    94→    if (step < STEPS.length - 1) {
    95→      setDirection(1);
    96→      setStep(step + 1);
    97→      scrollToWizard();
    98→    }
    99→  };
   100→
   101→  const back = () => {
   102→    if (step > 0) {
   103→      setDirection(-1);
   104→      setStep(step - 1);
   105→      scrollToWizard();
   106→    }
   107→  };
   108→
   109→  const submit = async () => {
   110→    setLoading(true);
   111→    try {
   112→      const res = await apiPost<Reservation>("/api/reservations", {
   113→        name: form.name.trim(),
   114→        phone: form.phone.trim(),
   115→        email: form.email.trim(),
   116→        date: form.date,
   117→        time: form.time,
   118→        guests: Number(form.guests),
   119→        special: form.special.trim(),
   120→      });
   121→      setSuccess(res);
   122→      toast.success("Reservation request received ✦");
   123→      scrollToWizard();
   124→    } catch (err) {
   125→      toast.error(err instanceof Error ? err.message : "Reservation failed");
   126→    } finally {
   127→      setLoading(false);
   128→    }
   129→  };
   130→
   131→  const reset = () => {
   132→    setSuccess(null);
   133→    setForm(EMPTY_FORM);
   134→    setStep(0);
   135→    setErrors({});
   136→    setDirection(1);
   137→    scrollToWizard();
   138→  };
   139→
   140→  return (
   141→    <div>
   142→      {/* ============== CINEMATIC HEADER ============== */}
   143→      <section className="relative flex min-h-[60vh] items-center justify-center overflow-hidden cinematic-grain">
   144→        <div className="absolute inset-0 -z-10">
   145→          <img src={IMAGES.ambiance[0]} alt="" loading="lazy" decoding="async" className="h-full w-full object-cover" />
   146→        </div>
   147→        <div className="absolute inset-0 bg-background/75" />
   148→        <div className="absolute inset-0 bg-gradient-to-b from-background/60 via-background/40 to-background" />
   149→        <div
   150→          className="absolute inset-0"
   151→          style={{ background: "radial-gradient(ellipse at center, transparent 35%, rgba(10,10,10,0.85) 100%)" }}
   152→        />
   153→
   154→        {/* Ambient gold orbs */}
   155→        <div className="ambient-orb" style={{ width: 420, height: 420, background: "rgba(212,175,55,0.14)", top: "14%", left: "6%" }} />
   156→        <div className="ambient-orb" style={{ width: 520, height: 520, background: "rgba(212,175,55,0.08)", bottom: "2%", right: "4%", animationDelay: "-5s" }} />
   157→
   158→        <div className="relative z-10 mx-auto max-w-4xl px-6 text-center">
   159→          <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} transition={{ duration: 0.8 }}>
   160→            <Eyebrow className="mb-6 justify-center">Reserve Your Evening</Eyebrow>
   161→          </motion.div>
   162→          <h1 className="font-[family-name:var(--font-playfair)] text-6xl font-semibold leading-[1.02] tracking-luxe text-foreground drop-shadow-[0_4px_30px_rgba(10,10,10,0.6)] sm:text-7xl lg:text-8xl">
   163→            <RevealText text="Online" as="span" delay={0.2} className="inline-block" />
   164→            <RevealText text="Reservation" as="span" delay={0.45} className="ml-3 inline-block text-gold-gradient sm:ml-5" />
   165→          </h1>
   166→          <OrnamentDivider className="mt-8" />
   167→          <motion.p
   168→            initial={{ opacity: 0, y: 16 }}
   169→            animate={{ opacity: 1, y: 0 }}
   170→            transition={{ delay: 0.9, duration: 0.8 }}
   171→            className="mx-auto mt-6 max-w-xl font-[family-name:var(--font-cormorant)] text-xl italic text-foreground/85 drop-shadow-[0_2px_16px_rgba(10,10,10,0.6)] sm:text-2xl"
   172→          >
   173→            Reserve in moments. Our maître d&apos; will confirm your table personally.
   174→          </motion.p>
   175→        </div>
   176→      </section>
   177→
   178→      {/* ============== WIZARD ============== */}
   179→      <section className="relative py-16 sm:py-24">
   180→        {/* Subtle ambient orbs in the section background */}
   181→        <div className="ambient-orb pointer-events-none absolute top-32 left-[-8%]" style={{ width: 360, height: 360, background: "rgba(212,175,55,0.05)" }} />

   183→

   204→
   205→                    {step === 1 && (

   210→                            <ArrowLeft className="h-4 w-4" /> Back
   211→                          </LuxuryButton>
   212→                          <LuxuryButton variant="solid" onClick={next} className="min-h-[52px] flex-1">
   213→                            Continue <ArrowRight className="h-4 w-4" />
   214→                          </LuxuryButton>
   215→                        </div>
   216→                      </motion.div>
   217→                    )}
   218→
   219→                    {step === 2 && (
   220→                      <motion.div key="step-2" custom={direction} variants={stepVariants} initial="enter" animate="center" exit="exit" transition={{ duration: 0.4, ease: [0.22, 1, 0.36, 1] }}>
   221→                        <StepGuests form={form} update={update} />
   222→                        <div className="mt-8 flex items-center gap-3">
   223→                          <LuxuryButton variant="ghost" onClick={back} className="min-h-[52px]">
   224→                            <ArrowLeft className="h-4 w-4" /> Back
   225→                          </LuxuryButton>
   226→                          <LuxuryButton variant="solid" onClick={next} className="min-h-[52px] flex-1">
   227→                            Continue <ArrowRight className="h-4 w-4" />
   228→                          </LuxuryButton>
   229→                        </div>
   230→                      </motion.div>
   231→                    )}
   232→
   233→                    {step === 3 && (
   234→                      <motion.div key="step-3" custom={direction} variants={stepVariants} initial="enter" animate="center" exit="exit" transition={{ duration: 0.4, ease: [0.22, 1, 0.36, 1] }}>
   235→                        <StepDetails form={form} errors={errors} update={update} />
   236→                        <div className="mt-8 flex items-center gap-3">
   237→                          <LuxuryButton variant="ghost" onClick={back} className="min-h-[52px]">
   238→                            <ArrowLeft className="h-4 w-4" /> Back
   239→                          </LuxuryButton>
   240→                          <LuxuryButton variant="solid" onClick={next} className="min-h-[52px] flex-1">

   242→                          </LuxuryButton>

   246→
   247→                    {step === 4 && (
   248→                      <motion.div key="step-4" custom={direction} variants={stepVariants} initial="enter" animate="center" exit="exit" transition={{ duration: 0.4, ease: [0.22, 1, 0.36, 1] }}>
   249→                        <StepConfirm form={form} update={update} />
   250→                        <div className="mt-8 flex items-center gap-3">

   258→                        <p className="mt-4 text-center font-sans text-[11px] text-muted-foreground/70">

   271→      <style>{`

   281→    </div>

   285→/* =========================================================
   286→   STEP INDICATOR — numbered circles + filling gold line
   287→   ========================================================= */
   288→function StepIndicator({ step }: { step: number }) {
   289→  const total = STEPS.length;
   290→  return (
   291→    <div className="mx-auto max-w-md">
   292→      <div className="relative flex items-start justify-between">
   293→        {/* Background hairline */}
   294→        <div className="absolute left-0 right-0 top-[18px] h-px bg-white/10 sm:top-[20px]" />
   295→        {/* Gold progress fill */}
   296→        <motion.div
   297→          className="absolute left-0 top-[18px] h-px bg-gold-gradient sm:top-[20px]"
   298→          initial={false}
   299→          animate={{ width: `${(step / (total - 1)) * 100}%` }}
   300→          transition={{ duration: 0.5, ease: "easeInOut" }}
   301→        />
   302→        {STEPS.map((label, i) => (
   303→          <div key={label} className="relative z-10 flex flex-col items-center">
   304→            <StepDot index={i} current={step} />
   305→            <span
   306→              className={cn(
   307→                "mt-2.5 font-sans text-[10px] uppercase tracking-[0.2em] transition-colors duration-300 sm:text-[11px]",
   308→                i === step ? "text-gold" : i < step ? "text-foreground/60" : "text-muted-foreground/50"
   309→              )}
   310→            >
   311→              {label}
   312→            </span>
   313→          </div>
   314→        ))}
   315→      </div>
   316→    </div>
   317→  );
   318→}
   319→
   320→function StepDot({ index, current }: { index: number; current: number }) {
   321→  const done = index < current;
   322→  const active = index === current;
   323→  return (
   324→    <div
   325→      className={cn(
   326→        "flex h-9 w-9 items-center justify-center rounded-full border-2 transition-all duration-300 sm:h-10 sm:w-10",
   327→        active && "border-gold bg-gold-gradient text-black glow-gold",
   328→        done && "border-gold bg-gold text-black",
   329→        !active && !done && "border-white/15 bg-background text-muted-foreground"
   330→      )}
   331→    >
   332→      {done ? (
   333→        <Check className="h-4 w-4 sm:h-5 sm:w-5" strokeWidth={3} />
   334→      ) : (
   335→        <span className="font-[family-name:var(--font-playfair)] text-sm font-semibold sm:text-base">
   336→          {index + 1}
   337→        </span>
   338→      )}
   339→    </div>
   340→  );
   341→}

   528→function GuestStepper({ value, onChange }: { value: string; onChange: (v: string) => void }) {
   529→  const n = Math.max(1, Math.min(MAX_GUESTS, Number(value) || 1));
   530→  const set = (v: number) => onChange(String(Math.max(1, Math.min(MAX_GUESTS, v))));

   533→      <div className="flex items-center gap-6 sm:gap-10">
   534→        <button
   535→          type="button"
   536→          onClick={() => set(n - 1)}
   537→          disabled={n <= 1}
   538→          aria-label="Decrease party size"
   539→          className="flex h-12 w-12 items-center justify-center rounded-full border border-white/10 text-foreground transition-all duration-200 hover:border-gold/50 hover:text-gold disabled:pointer-events-none disabled:opacity-30 sm:h-14 sm:w-14"

   580→          );
   581→        })}

   774→function formatDate(iso: string): string {
   775→  if (!iso) return "—";
   776→  const d = new Date(iso + "T00:00:00");
   777→  if (isNaN(d.getTime())) return iso;
   778→  return d.toLocaleDateString("en-US", {
   779→    weekday: "long", month: "long", day: "numeric", year: "numeric",
   780→  });
   781→}
