    52→  const wizardRef = useRef<HTMLDivElement>(null);
    53→
    54→  const today = new Date().toISOString().slice(0, 10);
    55→
    56→  const update = (key: keyof FormState, value: string) => {
    57→    setForm((f) => ({ ...f, [key]: value }));
    58→    setErrors((e) => (e[key] ? { ...e, [key]: "" } : e));
    59→  };
    60→
    61→  const validateStep = (s: number): Record<string, string> => {
    62→    const e: Record<string, string> = {};
    63→    if (s === 0) {
    64→      if (!form.name.trim()) e.name = "Please enter your name";
    65→      else if (form.name.trim().length < 2) e.name = "Name is too short";
    66→      if (!form.phone.trim()) e.phone = "Please enter your phone number";
    67→      else if (form.phone.replace(/\D/g, "").length < 7) e.phone = "Enter a valid phone number";
    68→      if (!form.email.trim()) e.email = "Please enter your email";
    69→      else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email.trim())) e.email = "Enter a valid email address";
    70→    } else if (s === 1) {
    71→      if (!form.date) e.date = "Please select a date";
    72→      if (!form.time) e.time = "Please select a time";
    73→    }
    74→    return e;
    75→  };

    80→      window.scrollTo({ top, behavior: "smooth" });
    81→    }
    82→  };
    83→
    84→  const next = () => {
    85→    const e = validateStep(step);
    86→    setErrors(e);
    87→    if (Object.keys(e).length > 0) return;
    88→    if (step < STEPS.length - 1) {
    89→      setDirection(1);
    90→      setStep(step + 1);
    91→      scrollToWizard();
    92→    }
    93→  };
    94→
    95→  const back = () => {

    97→      setDirection(-1);
    98→      setStep(step - 1);
    99→      scrollToWizard();
   100→    }
   101→  };

   104→    setLoading(true);

   108→        phone: form.phone.trim(),
   109→        email: form.email.trim(),
   110→        date: form.date,
   111→        time: form.time,
   112→        guests: Number(form.guests),
   113→        special: form.special.trim(),
   114→      });

   116→      toast.success("Reservation request received ✦");

   121→      setLoading(false);
   122→    }
   123→  };

   126→    setSuccess(null);
   127→    setForm(EMPTY_FORM);
   128→    setStep(0);
   129→    setErrors({});
   130→    setDirection(1);
   131→    scrollToWizard();
   132→  };
   133→
   134→  return (
   135→    <div>
   136→      {/* ============== CINEMATIC HEADER ============== */}
   137→      <section className="relative flex min-h-[60vh] items-center justify-center overflow-hidden cinematic-grain">
   138→        <div className="absolute inset-0 -z-10">
   139→          <img src={IMAGES.ambiance[0]} alt="" className="h-full w-full object-cover" />
   140→        </div>
   141→        <div className="absolute inset-0 bg-background/75" />
   142→        <div className="absolute inset-0 bg-gradient-to-b from-background/60 via-background/40 to-background" />
   143→        <div
   144→          className="absolute inset-0"
   145→          style={{ background: "radial-gradient(ellipse at center, transparent 35%, rgba(10,10,10,0.85) 100%)" }}
   146→        />
   147→
   148→        {/* Ambient gold orbs */}
   149→        <div className="ambient-orb" style={{ width: 420, height: 420, background: "rgba(212,175,55,0.14)", top: "14%", left: "6%" }} />
   150→        <div className="ambient-orb" style={{ width: 520, height: 520, background: "rgba(212,175,55,0.08)", bottom: "2%", right: "4%", animationDelay: "-5s" }} />
   151→
   152→        <div className="relative z-10 mx-auto max-w-4xl px-6 text-center">
   153→          <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} transition={{ duration: 0.8 }}>
   154→            <Eyebrow className="mb-6 justify-center">Reserve Your Evening</Eyebrow>
   155→          </motion.div>
   156→          <h1 className="font-[family-name:var(--font-playfair)] text-6xl font-semibold leading-[1.02] tracking-luxe text-foreground drop-shadow-[0_4px_30px_rgba(10,10,10,0.6)] sm:text-7xl lg:text-8xl">

   164→            transition={{ delay: 0.9, duration: 0.8 }}
   165→            className="mx-auto mt-6 max-w-xl font-[family-name:var(--font-cormorant)] text-xl italic text-foreground/85 drop-shadow-[0_2px_16px_rgba(10,10,10,0.6)] sm:text-2xl"

   207→                    {step === 1 && (

   231→                        key="step-2"
   232→                        custom={direction}
   233→                        variants={stepVariants}

   236→                        exit="exit"
   237→                        transition={{ duration: 0.4, ease: [0.22, 1, 0.36, 1] }}
   238→                      >
   239→                        <Step3Review form={form} update={update} />
   240→                        <div className="mt-8 flex items-center gap-3">

   286→/* =========================================================
   287→   STEP INDICATOR — numbered circles + filling gold line
   288→   ========================================================= */
   289→function StepIndicator({ step }: { step: number }) {
   290→  const total = STEPS.length;
   291→  return (
   292→    <div className="mx-auto max-w-md">
   293→      <div className="relative flex items-start justify-between">
   294→        {/* Background hairline */}
   295→        <div className="absolute left-0 right-0 top-[18px] h-px bg-white/10 sm:top-[20px]" />
   296→        {/* Gold progress fill */}
   297→        <motion.div
   298→          className="absolute left-0 top-[18px] h-px bg-gold-gradient sm:top-[20px]"
   299→          initial={false}
   300→          animate={{ width: `${(step / (total - 1)) * 100}%` }}
   301→          transition={{ duration: 0.5, ease: "easeInOut" }}
   302→        />
   303→        {STEPS.map((label, i) => (
   304→          <div key={label} className="relative z-10 flex flex-col items-center">
   305→            <StepDot index={i} current={step} />
   306→            <span
   307→              className={cn(
   308→                "mt-2.5 font-sans text-[10px] uppercase tracking-[0.2em] transition-colors duration-300 sm:text-[11px]",
   309→                i === step ? "text-gold" : i < step ? "text-foreground/60" : "text-muted-foreground/50"
   310→              )}
   311→            >
   312→              {label}
   313→            </span>
   314→          </div>
   315→        ))}
   316→      </div>
   317→    </div>
   318→  );
   319→}
   320→
   321→function StepDot({ index, current }: { index: number; current: number }) {
   322→  const done = index < current;
   323→  const active = index === current;
   324→  return (
   325→    <div
   326→      className={cn(
   327→        "flex h-9 w-9 items-center justify-center rounded-full border-2 transition-all duration-300 sm:h-10 sm:w-10",
   328→        active && "border-gold bg-gold-gradient text-black glow-gold",
   329→        done && "border-gold bg-gold text-black",
   330→        !active && !done && "border-white/15 bg-background text-muted-foreground"
   331→      )}
   332→    >
   333→      {done ? (
   334→        <Check className="h-4 w-4 sm:h-5 sm:w-5" strokeWidth={3} />
   335→      ) : (
   336→        <span className="font-[family-name:var(--font-playfair)] text-sm font-semibold sm:text-base">
   337→          {index + 1}
   338→        </span>
   339→      )}
   340→    </div>
   341→  );
   342→}

   495→function GuestStepper({ value, onChange }: { value: string; onChange: (v: string) => void }) {
   496→  const n = Math.max(1, Math.min(MAX_GUESTS, Number(value) || 1));
   497→  const set = (v: number) => onChange(String(Math.max(1, Math.min(MAX_GUESTS, v))));
   498→  return (
   499→    <div className="flex flex-col items-center">
   500→      <div className="flex items-center gap-6 sm:gap-10">
   501→        <button
   502→          type="button"
   503→          onClick={() => set(n - 1)}
   504→          disabled={n <= 1}
   505→          aria-label="Decrease party size"
   506→          className="flex h-12 w-12 items-center justify-center rounded-full border border-white/10 text-foreground transition-all duration-200 hover:border-gold/50 hover:text-gold disabled:pointer-events-none disabled:opacity-30 sm:h-14 sm:w-14"
   507→        >
   508→          <Minus className="h-4 w-4" />
   509→        </button>
   510→        <div className="flex min-w-[88px] flex-col items-center">
   511→          <span className="font-[family-name:var(--font-playfair)] text-5xl font-semibold text-gold-gradient sm:text-6xl">
   512→            {n}
   513→          </span>
   514→          <span className="mt-1 font-sans text-[10px] uppercase tracking-[0.25em] text-muted-foreground">
   515→            {n === 1 ? "Guest" : "Guests"}
   516→          </span>
   517→        </div>
   518→        <button
   519→          type="button"
   520→          onClick={() => set(n + 1)}
   521→          disabled={n >= MAX_GUESTS}
   522→          aria-label="Increase party size"

   786→function formatDate(iso: string): string {
   787→  if (!iso) return "—";
   788→  const d = new Date(iso + "T00:00:00");
   789→  if (isNaN(d.getTime())) return iso;
   790→  return d.toLocaleDateString("en-US", {
   791→    weekday: "long", month: "long", day: "numeric", year: "numeric",
   792→  });
   793→}
