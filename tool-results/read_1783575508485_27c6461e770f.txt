    59→/* =========================================================
    60→   MULTI-IMAGE UPLOADER
    61→   - Grid of thumbnails with cover badge on index 0
    62→   - Hover overlay: Replace / Delete + Left/Right reorder
    63→   - "Add Image" tile (file picker → data URL)
    64→   - Paste URL fallback
    65→   - image/* only, 6MB max
    66→   ========================================================= */
    67→function MultiImageUploader({
    68→  value, onChange,
    69→}: {
    70→  value: string[];
    71→  onChange: (v: string[]) => void;
    72→}) {
    73→  const [error, setError] = useState("");
    74→  const [urlInput, setUrlInput] = useState("");
    75→  const fileRef = useRef<HTMLInputElement | null>(null);
    76→  // When set, the next file pick replaces this index instead of appending.
    77→  const replaceIndexRef = useRef<number | null>(null);
    78→
    79→  const validate = (file: File) => {
    80→    const okType = /image\/(jpeg|jpg|png|webp|gif|avif)/.test(file.type);
    81→    const okSize = file.size <= 6 * 1024 * 1024;
    82→    if (!okType) { setError("Only JPG, PNG, WebP, GIF, AVIF allowed"); return false; }
    83→    if (!okSize) { setError("Max file size is 6MB"); return false; }
    84→    setError("");
    85→    return true;
    86→  };
    87→
    88→  const readFile = (file: File, cb: (dataUrl: string) => void) => {
    89→    const reader = new FileReader();
    90→    reader.onload = () => cb(reader.result as string);
    91→    reader.onerror = () => setError("Could not read file");
    92→    reader.readAsDataURL(file);
    93→  };
    94→
    95→  const onPick = (file?: File) => {
    96→    if (!file) return;
    97→    if (!validate(file)) return;
    98→    readFile(file, (dataUrl) => {
    99→      if (replaceIndexRef.current !== null) {
   100→        const idx = replaceIndexRef.current;
   101→        const next = [...value];
   102→        next[idx] = dataUrl;
   103→        onChange(next);
   104→        replaceIndexRef.current = null;
   105→      } else {
   106→        onChange([...value, dataUrl]);
   107→      }
   108→    });
   109→  };
   110→
   111→  const triggerAdd = () => {
   112→    replaceIndexRef.current = null;
   113→    fileRef.current?.click();
   114→  };
   115→
   116→  const triggerReplace = (idx: number) => {
   117→    replaceIndexRef.current = idx;
   118→    fileRef.current?.click();
   119→  };
   120→
   121→  const remove = (idx: number) => {
   122→    onChange(value.filter((_, i) => i !== idx));
   123→  };
   124→
   125→  const move = (idx: number, dir: -1 | 1) => {
   126→    const target = idx + dir;
   127→    if (target < 0 || target >= value.length) return;
   128→    const next = [...value];
   129→    [next[idx], next[target]] = [next[target], next[idx]];
   130→    onChange(next);
   131→  };
   132→
   133→  const addUrl = () => {
   134→    const u = urlInput.trim();
   135→    if (!u) return;
   136→    setError("");
   137→    onChange([...value, u]);
   138→    setUrlInput("");
   139→  };
   140→
   141→  return (
   142→    <div className="block">
   143→      <div className="mb-1.5 flex items-center justify-between">
   144→        <span className="admin-label">Dish Images</span>
   145→        <span className="font-sans text-[11px] text-admin-muted">
   146→          {value.length} image{value.length !== 1 ? "s" : ""} · first is the cover
   147→        </span>
   148→      </div>
   149→
   150→      <div className="grid grid-cols-3 gap-3 sm:grid-cols-4">
   151→        {value.map((img, idx) => (
   152→          <div
   153→            key={`${idx}-${img.slice(0, 32)}`}
   154→            className="group relative aspect-square overflow-hidden rounded-xl border border-admin-border bg-admin-bg"
   155→          >
   156→            <img src={img} alt={`Dish image ${idx + 1}`} className="h-full w-full object-cover" />
   157→            {idx === 0 && (
   158→              <span className="absolute left-1.5 top-1.5 inline-flex items-center gap-1 rounded-full border border-admin-gold/40 bg-admin-gold/95 px-1.5 py-0.5 font-sans text-[9px] font-bold uppercase tracking-wider text-black">
   159→                <Star className="h-2 w-2 fill-black text-black" /> Cover
   160→              </span>
   161→            )}
   162→            <span className="absolute right-1.5 top-1.5 rounded-full bg-black/55 px-1.5 py-0.5 font-sans text-[9px] font-medium text-white/80 backdrop-blur-sm">
   163→              {idx + 1}
   164→            </span>
   165→
   166→            {/* Hover overlay */}
   167→            <div className="absolute inset-0 flex flex-col items-center justify-center gap-1.5 bg-black/70 opacity-0 backdrop-blur-sm transition-opacity duration-200 group-hover:opacity-100">
   168→              <div className="flex gap-1">
   169→                <button
   170→                  type="button"
   171→                  onClick={() => triggerReplace(idx)}
   172→                  aria-label={`Replace image ${idx + 1}`}
   173→                  className="flex h-8 w-8 items-center justify-center rounded-lg border border-white/15 bg-white/10 text-white transition-colors hover:border-admin-gold/50 hover:bg-admin-gold/20 hover:text-admin-gold"
   174→                >
   175→                  <Upload className="h-3.5 w-3.5" />
   176→                </button>
   177→                <button
   178→                  type="button"
   179→                  onClick={() => remove(idx)}
   180→                  aria-label={`Delete image ${idx + 1}`}
   181→                  className="flex h-8 w-8 items-center justify-center rounded-lg border border-red-500/30 bg-red-500/10 text-red-300 transition-colors hover:bg-red-500/20 hover:text-red-200"
   182→                >
   183→                  <Trash2 className="h-3.5 w-3.5" />
   184→                </button>
   185→              </div>
   186→              <div className="flex gap-1">
   187→                <button
   188→                  type="button"
   189→                  onClick={() => move(idx, -1)}
   190→                  disabled={idx === 0}
   191→                  aria-label="Move image left"
   192→                  className="flex h-7 w-7 items-center justify-center rounded-lg border border-white/15 bg-white/10 text-white transition-colors hover:border-admin-gold/50 hover:bg-admin-gold/20 hover:text-admin-gold disabled:opacity-30 disabled:hover:bg-white/10 disabled:hover:text-white"
   193→                >
   194→                  <ChevronLeft className="h-3.5 w-3.5" />
   195→                </button>
   196→                <button
   197→                  type="button"
   198→                  onClick={() => move(idx, 1)}
   199→                  disabled={idx === value.length - 1}
   200→                  aria-label="Move image right"
   201→                  className="flex h-7 w-7 items-center justify-center rounded-lg border border-white/15 bg-white/10 text-white transition-colors hover:border-admin-gold/50 hover:bg-admin-gold/20 hover:text-admin-gold disabled:opacity-30 disabled:hover:bg-white/10 disabled:hover:text-white"
   202→                >
   203→                  <ChevronRight className="h-3.5 w-3.5" />
   204→                </button>
   205→              </div>
   206→            </div>
   207→          </div>
   208→        ))}
   209→
   210→        {/* Add tile */}
   211→        <button
   212→          type="button"
   213→          onClick={triggerAdd}
   214→          className="flex aspect-square flex-col items-center justify-center gap-2 rounded-xl border-2 border-dashed border-admin-border bg-admin-bg/40 text-admin-muted transition-all duration-200 hover:border-admin-gold/40 hover:bg-admin-gold/5 hover:text-admin-gold"
   215→        >
   216→          <Upload className="h-5 w-5" />
   217→          <span className="font-sans text-[10px] font-semibold uppercase tracking-wider">Add Image</span>
   218→        </button>
   219→      </div>
   220→
   221→      {/* URL fallback */}
   222→      <div className="mt-3 flex gap-2">
   223→        <input
   224→          value={urlInput}
   225→          onChange={(e) => setUrlInput(e.target.value)}
   226→          onKeyDown={(e) => {
   227→            if (e.key === "Enter") {
   228→              e.preventDefault();
   229→              addUrl();
   230→            }
   231→          }}
   232→          placeholder="Or paste an image URL…"
   233→          className="admin-input flex-1"
   234→        />
   235→        <AdminButton
   236→          type="button"
   237→          variant="subtle"
   238→          onClick={addUrl}
   239→          disabled={!urlInput.trim()}
   240→        >
   241→          <LinkIcon className="h-3.5 w-3.5" /> Add URL
   242→        </AdminButton>
   243→      </div>
   244→
   245→      <p className="mt-1.5 font-sans text-xs text-admin-muted">
   246→        JPG, PNG, WebP, GIF, AVIF · up to 6MB each
   247→      </p>
   248→
   249→      {/* Hidden file input shared by Add + Replace */}
   250→      <input
   251→        ref={fileRef}
   252→        type="file"
   253→        accept="image/*"
   254→        className="hidden"
   255→        onChange={(e) => {
   256→          const f = e.target.files?.[0];
   257→          if (f) onPick(f);
   258→          e.target.value = "";
   259→        }}
   260→      />
   261→
   262→      {error && (
   263→        <p className="mt-1.5 flex items-center gap-1 font-sans text-xs text-red-400">
   264→          <AlertTriangle className="h-3 w-3" />{error}
   265→        </p>
   266→      )}
   267→    </div>
   268→  );
   269→}

   271→/* =========================================================
   272→   TAG INPUT — Linear/Notion-style chip editor
   273→   Used for ingredients (gold) and allergens (red)
   274→   ========================================================= */
   275→function TagInput({
   276→  label, values, onChange, placeholder, tone = "neutral", suggestions,
   277→}: {
   278→  label: string;
   279→  values: string[];
   280→  onChange: (v: string[]) => void;
   281→  placeholder?: string;
   282→  tone?: "neutral" | "red";
   283→  suggestions?: string[];
   284→}) {
   285→  const [input, setInput] = useState("");
   286→
   287→  const add = (raw: string) => {
   288→    const v = raw.trim();
   289→    if (!v) return;
   290→    if (values.some((x) => x.toLowerCase() === v.toLowerCase())) {
   291→      setInput("");
   292→      return;
   293→    }
   294→    onChange([...values, v]);
   295→    setInput("");
   296→  };
   297→
   298→  const onKey = (e: React.KeyboardEvent<HTMLInputElement>) => {
   299→    if (e.key === "Enter" || e.key === ",") {
   300→      e.preventDefault();
   301→      add(input);
   302→    }
   303→    if (e.key === "Backspace" && !input && values.length > 0) {
   304→      onChange(values.slice(0, -1));
   305→    }
   306→  };
   307→
   308→  const remove = (idx: number) => {
   309→    onChange(values.filter((_, i) => i !== idx));
   310→  };
   311→
   312→  const isRed = tone === "red";
   313→  const chipClass = isRed
   314→    ? "border-red-500/30 bg-red-500/10 text-red-300"
   315→    : "border-admin-gold/30 bg-admin-gold/10 text-admin-gold";
   316→  const xClass = isRed
   317→    ? "text-red-400/60 hover:text-red-200"
   318→    : "text-admin-gold/60 hover:text-admin-gold";
   319→  const containerClass = isRed
   320→    ? "border-red-500/20 focus-within:border-red-500/40"
   321→    : "border-admin-border focus-within:border-admin-gold/40";
   322→  const suggestionClass = isRed
   323→    ? "border-red-500/20 text-red-400/80 hover:border-red-500/40 hover:bg-red-500/10 hover:text-red-300"
   324→    : "border-admin-border text-admin-muted hover:border-admin-gold/40 hover:bg-admin-gold/10 hover:text-admin-gold";
   325→
   326→  const availableSuggestions =
   327→    suggestions?.filter(
   328→      (s) => !values.some((v) => v.toLowerCase() === s.toLowerCase())
   329→    ) ?? [];
   330→
   331→  return (
   332→    <div className="block">
   333→      <span className="mb-1.5 flex items-center gap-1 admin-label">{label}</span>
   334→      <div
   335→        className={`rounded-xl border bg-white/[0.02] p-2 transition-colors ${containerClass}`}
   336→      >
   337→        <div className="flex flex-wrap items-center gap-1.5">
   338→          {values.map((v, idx) => (
   339→            <span
   340→              key={`${idx}-${v}`}
   341→              className={`inline-flex items-center gap-1 rounded-lg border px-2 py-1 font-sans text-xs ${chipClass}`}
   342→            >
   343→              {v}
   344→              <button
   345→                type="button"
   346→                onClick={() => remove(idx)}
   347→                aria-label={`Remove ${v}`}
   348→                className={`transition-colors ${xClass}`}
   349→              >
   350→                <X className="h-3 w-3" />
   351→              </button>
   352→            </span>
   353→          ))}
   354→          <input
   355→            value={input}
   356→            onChange={(e) => setInput(e.target.value)}
   357→            onKeyDown={onKey}
   358→            onBlur={() => input && add(input)}
   359→            placeholder={values.length === 0 ? placeholder : "Add another…"}
   360→            className="min-w-[120px] flex-1 bg-transparent px-1 py-1 font-sans text-sm text-admin-text outline-none placeholder:text-admin-muted/60"
   361→          />
   362→        </div>
   363→      </div>
   364→      {availableSuggestions.length > 0 && (
   365→        <div className="mt-2 flex flex-wrap items-center gap-1.5">
   366→          <span className="font-sans text-[10px] uppercase tracking-wider text-admin-muted">
   367→            Quick add:
   368→          </span>
   369→          {availableSuggestions.map((s) => (
   370→            <button
   371→              key={s}
   372→              type="button"
   373→              onClick={() => add(s)}
   374→              className={`rounded-md border px-1.5 py-0.5 font-sans text-[11px] transition-colors ${suggestionClass}`}
   375→            >
   376→              + {s}
   377→            </button>
   378→          ))}
   379→        </div>
   380→      )}
   381→    </div>
   382→  );
   383→}

   515→                            <p className="mt-0.5 truncate font-sans text-xs text-admin-muted">{item.description}</p>
   516→                          ) : null}
   517→                        </div>
   518→                        <span className="shrink-0 font-[family-name:var(--font-playfair)] text-lg font-semibold text-admin-gold">${item.price}</span>
   519→                        <div className="flex shrink-0 gap-1 opacity-0 transition-opacity duration-200 focus-within:opacity-100 group-hover:opacity-100">
   520→                          <button
   521→                            onClick={() => setItemModal(item)}
   522→                            aria-label={`Edit ${item.name}`}
   523→                            className="flex h-9 w-9 items-center justify-center rounded-lg border border-admin-gold/30 text-admin-gold transition-colors hover:bg-admin-gold/10"
   524→                          >
   525→                            <Pencil className="h-3.5 w-3.5" />
   526→                          </button>
   527→                          <button
   528→                            onClick={() => {
   529→                              apiDelete(`/api/menu/${item.id}`)
   530→                                .then(() => { toast.success("Item deleted"); load(); })
   531→                                .catch(() => toast.error("Delete failed"));
   532→                            }}
   533→                            aria-label={`Delete ${item.name}`}
   534→                            className="flex h-9 w-9 items-center justify-center rounded-lg border border-red-500/30 text-red-400 transition-colors hover:bg-red-500/10"
   535→                          >
   536→                            <Trash2 className="h-3.5 w-3.5" />
   537→                          </button>
   538→                        </div>
   539→                      </div>
   540→                    );
   541→                  })}
   542→                </div>
   543→              )}
   544→            </AdminCard>
   545→          ))}
   546→        </div>
   547→      )}
   548→
   549→      {/* Item modal — conditionally mounted so form state initialises cleanly per item */}
   550→      {itemModal !== null && (
   551→        <ItemModal
   552→          key={editingItem?.id ?? "new"}
   553→          item={editingItem}
   554→          categories={categories ?? []}
   555→          onClose={() => setItemModal(null)}
   556→          onSaved={() => { setItemModal(null); load(); }}
   557→        />
   558→      )}
   559→
   560→      {/* Category modal */}
   561→      {catModal && (
   562→        <CategoryModal
   563→          onClose={() => setCatModal(false)}
   564→          onSaved={() => { setCatModal(false); load(); }}
   565→        />
   566→      )}
   567→    </div>
   568→  );
   569→}

   571→function CategoryActions({ cat, onChanged }: { cat: MenuCategory; onChanged: () => void }) {
   572→  const rename = () => {
   573→    const name = window.prompt("Rename category", cat.name);
   574→    if (name && name.trim()) {
   575→      apiPatch(`/api/categories/${cat.id}`, { name: name.trim() })
   576→        .then(onChanged)
   577→        .then(() => toast.success("Category renamed"))
   578→        .catch(() => toast.error("Rename failed"));
   579→    }
   580→  };
   581→  const remove = () => {
   582→    apiDelete(`/api/categories/${cat.id}`)
   583→      .then(onChanged)
   584→      .then(() => toast.success("Category deleted"))
   585→      .catch(() => toast.error("Delete failed"));
   586→  };
   587→  return (
   588→    <div className="flex shrink-0 gap-1">
   589→      <button
   590→        onClick={rename}
   591→        aria-label={`Rename ${cat.name}`}
   592→        className="flex h-9 w-9 items-center justify-center rounded-lg border border-admin-gold/30 text-admin-gold transition-colors hover:bg-admin-gold/10"
   593→      >
   594→        <Pencil className="h-3.5 w-3.5" />
   595→      </button>
   596→      <button
   597→        onClick={() => {
   598→          if (window.confirm(`Delete category "${cat.name}" and all its items? This cannot be undone.`)) remove();
   599→        }}
   600→        aria-label={`Delete ${cat.name}`}
   601→        className="flex h-9 w-9 items-center justify-center rounded-lg border border-red-500/30 text-red-400 transition-colors hover:bg-red-500/10"
   602→      >
   603→        <Trash2 className="h-3.5 w-3.5" />
   604→      </button>
   605→    </div>
   606→  );

   627→function deriveItemForm(item: MenuItem | null, categories: MenuCategory[]): ItemFormState {
   628→  if (item) {
   629→    const imgs = item.images && item.images.length > 0
   630→      ? item.images
   631→      : item.image ? [item.image] : [];
   632→    return {
   633→      name: item.name,
   634→      tagline: item.tagline ?? "",
   635→      shortDescription: item.shortDescription ?? "",
   636→      description: item.description,
   637→      price: String(item.price),
   638→      images: imgs,
   639→      categoryId: item.categoryId,

   642→      spice: String(item.spice),
   643→      featured: item.featured,
   644→      chefRecommended: item.chefRecommended ?? false,
   645→      ingredients: item.ingredients ?? [],
   646→      allergens: item.allergens ?? [],
   647→      servingSize: item.servingSize ?? "",
   648→    };
   649→  }
   650→  return {
   651→    name: "",
   652→    tagline: "",
   653→    shortDescription: "",
   654→    description: "",
   655→    price: "",
   656→    images: [],
   657→    categoryId: categories[0]?.id ?? "",
   658→    available: true,
   659→    veg: false,

   661→    featured: false,

   665→    servingSize: "",
   666→  };

   669→function ItemModal({
   670→  item, categories, onClose, onSaved,
   671→}: {
   672→  item: MenuItem | null;
   673→  categories: MenuCategory[];
   674→  onClose: () => void;
   675→  onSaved: () => void;
   676→}) {
   677→  // Lazy initialiser + conditional mount via `key` (in parent) keeps this lint-safe
   678→  // (no setState-in-effect). Form resets cleanly per item.
   679→  const [form, setForm] = useState<ItemFormState>(() => deriveItemForm(item, categories));
   680→  const [saving, setSaving] = useState(false);
   681→
   682→  const set = <K extends keyof ItemFormState>(k: K, v: ItemFormState[K]) =>
   683→    setForm((s) => ({ ...s, [k]: v }));
   684→
   685→  const save = async () => {
   686→    if (!form.name.trim() || !form.price || !form.categoryId) return;
   687→    setSaving(true);
   688→    try {
   689→      const payload = {
   690→        name: form.name.trim(),
   691→        tagline: form.tagline.trim(),
   692→        shortDescription: form.shortDescription.trim(),
   693→        description: form.description,
   694→        price: Number(form.price),
   695→        image: form.images[0] ?? null,
   696→        images: form.images,
   697→        categoryId: form.categoryId,
   698→        available: form.available,
   699→        veg: form.veg,
   700→        spice: Number(form.spice),
   701→        featured: form.featured,
   702→        chefRecommended: form.chefRecommended,
   703→        ingredients: form.ingredients,
   704→        allergens: form.allergens,
   705→        servingSize: form.servingSize.trim(),
   706→      };
   707→      if (item) await apiPatch(`/api/menu/${item.id}`, payload);
   708→      else await apiPost("/api/menu", payload);
   709→      toast.success(item ? "Item updated" : "Item created");
   710→      onSaved();
   711→    } catch (e) {
   712→      toast.error(e instanceof Error ? e.message : "Save failed");
   713→    } finally {
   714→      setSaving(false);
   715→    }
   716→  };
   717→
   718→  const canSave =
   719→    !saving && form.name.trim().length > 0 && form.price !== "" && form.categoryId !== "";
   720→
   721→  return (
   722→    <Modal
   723→      open
   724→      onClose={onClose}
   725→      title={item ? "Edit Menu Item" : "Add Menu Item"}
   726→      subtitle={item ? item.name : "Create a new dish on the menu"}
   727→      size="xl"
   728→      footer={
   729→        <>
   730→          <AdminButton variant="ghost" onClick={onClose}>Cancel</AdminButton>
   731→          <AdminButton
   732→            variant="solid"
   733→            onClick={save}
   734→            disabled={!canSave}
   735→          >
   736→            {saving ? "Saving…" : "Save Item"}
   737→          </AdminButton>
   738→        </>
   739→      }
   740→    >
   741→      <div className="space-y-7">
   742→        {/* ───────────── Essentials ───────────── */}
   743→        <section className="space-y-4">
   744→          <SectionLabel>Essentials</SectionLabel>
   745→          <AdminInput
   746→            label="Name"
   747→            required
   748→            value={form.name}
   749→            onChange={(e) => set("name", e.target.value)}
   750→            placeholder="e.g. Wagyu Tenderloin"
   751→          />
   752→          <AdminInput
   753→            label="Tagline"
   754→            value={form.tagline}
   755→            onChange={(e) => set("tagline", e.target.value)}
   756→            placeholder="A short evocative line — e.g. The crown jewel of the menu"
   757→            hint="Shown beneath the dish name on the menu"
   758→          />
   759→          <div className="grid gap-4 md:grid-cols-2">
   760→            <SearchableSelect
   761→              label="Category"
   762→              required
   763→              value={form.categoryId}
   764→              onChange={(v) => set("categoryId", v)}
   765→              options={categories.map((c) => ({ value: c.id, label: c.name }))}
   766→              placeholder="Select a category"
   767→            />
   768→            <AdminInput
   769→              label="Price"
   770→              required
   771→              type="number"
   772→              step="0.01"
   773→              min="0"
   774→              value={form.price}
   775→              onChange={(e) => set("price", e.target.value)}
   776→              placeholder="0.00"
   777→            />
   778→          </div>
   779→          <AdminInput
   780→            label="Short Description"
   781→            value={form.shortDescription}
   782→            onChange={(e) => set("shortDescription", e.target.value)}
   783→            placeholder="One-line summary for menu lists"
   784→            hint="A single line used in compact menu layouts"
   785→          />
   786→          <AdminTextarea
   787→            label="Full Description"
   788→            required
   789→            value={form.description}
   790→            onChange={(e) => set("description", e.target.value)}
   791→            rows={4}
   792→            placeholder="A rich, evocative description of the dish"
   793→          />
   794→        </section>
   795→
   796→        {/* ───────────── Imagery ───────────── */}
   797→        <section className="space-y-4">
   798→          <SectionLabel>Imagery</SectionLabel>
   799→          <MultiImageUploader

   808→          <div className="grid gap-4 md:grid-cols-2">
   809→            <SearchableSelect
   810→              label="Spice Level"
   811→              value={form.spice}
   812→              onChange={(v) => set("spice", v)}
   813→              options={SPICE_OPTIONS}
   814→              searchable={false}
   815→            />
   816→            <AdminInput
   817→              label="Serving Size"
   818→              value={form.servingSize}
   819→              onChange={(e) => set("servingSize", e.target.value)}
   820→              placeholder="e.g. 200g, 4 pieces"
   821→            />

   826→            <Toggle label="Chef's Recommendation" color="gold" checked={form.chefRecommended} onChange={(v) => set("chefRecommended", v)} />
   827→            <Toggle label="Available" color="green" checked={form.available} onChange={(v) => set("available", v)} />
   828→          </div>
   829→        </section>
   830→
   831→        {/* ───────────── Ingredients & Allergens ───────────── */}
   832→        <section className="space-y-4">
   833→          <SectionLabel>Ingredients & Allergens</SectionLabel>
   834→          <TagInput

   839→            tone="neutral"

   841→          <TagInput
   842→            label="Allergens"

   855→function CategoryModal({ onClose, onSaved }: { onClose: () => void; onSaved: () => void }) {

   860→    if (!name.trim()) return;

   863→      await apiPost("/api/menu", { _type: "category", name: name.trim() });
   864→      toast.success("Category created");
   865→      onSaved();
   866→    } catch (e) {
   867→      toast.error(e instanceof Error ? e.message : "Failed to create category");

   869→      setSaving(false);

   875→      open
   876→      onClose={onClose}
