    69→export function Sparkline({ data, color = "#d4af37" }: { data: number[]; color?: string }) {
    70→  const max = Math.max(...data, 1);
    71→  const min = Math.min(...data, 0);
    72→  const range = max - min || 1;
    73→  const pts = data.map((v, i) => {
    74→    const x = (i / (data.length - 1)) * 100;
    75→    const y = 100 - ((v - min) / range) * 100;
    76→    return `${x},${y}`;
    77→  }).join(" ");
    78→  return (
    79→    <svg viewBox="0 0 100 100" preserveAspectRatio="none" className="h-full w-full">
    80→      <defs>
    81→        <linearGradient id={`spark-${color.replace("#", "")}`} x1="0" y1="0" x2="0" y2="1">
    82→          <stop offset="0%" stopColor={color} stopOpacity="0.35" />
    83→          <stop offset="100%" stopColor={color} stopOpacity="0" />
    84→        </linearGradient>
    85→      </defs>
    86→      <polygon points={`0,100 ${pts} 100,100`} fill={`url(#spark-${color.replace("#", "")})`} />
    87→      <polyline points={pts} fill="none" stroke={color} strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" vectorEffect="non-scaling-stroke" />
    88→    </svg>
    89→  );
    90→}

   107→/* =========================================================
   108→   MODAL — premium, fast, accessible
   109→   - Opens instantly (150ms fade + scale)
   110→   - 70% dark overlay + backdrop blur
   111→   - z-[100] above all content
   112→   - Body scroll lock
   113→   - Escape to close
   114→   - Focus trap + focus restoration
   115→   - ARIA dialog attributes
   116→   - Sticky footer with Cancel/Save
   117→   ========================================================= */
   118→export function Modal({
   119→  open, onClose, title, subtitle, children, footer, wide = false, size = "md",
   120→}: {
   121→  open: boolean; onClose: () => void; title: string; subtitle?: string;
   122→  children: React.ReactNode; footer?: React.ReactNode; wide?: boolean; size?: "sm" | "md" | "lg" | "xl";
   123→}) {
   124→  const dialogRef = useRef<HTMLDivElement | null>(null);
   125→  const previouslyFocused = useRef<HTMLElement | null>(null);
   126→
   127→  // Body scroll lock
   128→  useEffect(() => {
   129→    if (open) {
   130→      document.body.style.overflow = "hidden";
   131→      return () => { document.body.style.overflow = ""; };
   132→    }
   133→    document.body.style.overflow = "";
   134→  }, [open]);
   135→
   136→  // Escape to close + focus trap
   137→  useEffect(() => {
   138→    if (!open) return;
   139→    previouslyFocused.current = document.activeElement as HTMLElement;
   140→
   141→    const onKeyDown = (e: KeyboardEvent) => {
   142→      if (e.key === "Escape") { e.preventDefault(); onClose(); return; }
   143→      if (e.key === "Tab") {
   144→        const dialog = dialogRef.current;
   145→        if (!dialog) return;
   146→        const focusable = dialog.querySelectorAll<HTMLElement>(
   147→          'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])'
   148→        );
   149→        if (focusable.length === 0) return;
   150→        const first = focusable[0];
   151→        const last = focusable[focusable.length - 1];
   152→        if (e.shiftKey && document.activeElement === first) {
   153→          e.preventDefault(); last.focus();
   154→        } else if (!e.shiftKey && document.activeElement === last) {
   155→          e.preventDefault(); first.focus();
   156→        }
   157→      }
   158→    };
   159→    document.addEventListener("keydown", onKeyDown);
   160→
   161→    // Focus the dialog (or first focusable) on open — microtask delay so DOM is ready
   162→    const t = setTimeout(() => {
   163→      const dialog = dialogRef.current;
   164→      if (!dialog) return;
   165→      const first = dialog.querySelector<HTMLElement>('input, button, select, textarea, [tabindex]:not([tabindex="-1"])');
   166→      (first || dialog).focus();
   167→    }, 50);
   168→
   169→    // Restore focus to the trigger on close
   170→    return () => {
   171→      document.removeEventListener("keydown", onKeyDown);
   172→      clearTimeout(t);
   173→      previouslyFocused.current?.focus?.();
   174→    };
   175→  }, [open, onClose]);
   176→
   177→  const maxW = wide ? "max-w-2xl" : size === "xl" ? "max-w-4xl" : size === "lg" ? "max-w-2xl" : size === "sm" ? "max-w-sm" : "max-w-lg";
   178→
   179→  return (
   180→    <AnimatePresence>
   181→      {open && (
   182→        <motion.div
   183→          initial={{ opacity: 0 }}
   184→          animate={{ opacity: 1 }}
   185→          exit={{ opacity: 0 }}
   186→          transition={{ duration: 0.15 }}
   187→          className="fixed inset-0 z-[100] flex items-start justify-center overflow-y-auto bg-black/70 p-4 backdrop-blur-md sm:p-8"
   188→          onClick={onClose}
   189→          role="presentation"
   190→        >
   191→          <motion.div
   192→            ref={dialogRef}
   193→            initial={{ opacity: 0, y: 12, scale: 0.97 }}
   194→            animate={{ opacity: 1, y: 0, scale: 1 }}
   195→            exit={{ opacity: 0, y: 8, scale: 0.98 }}
   196→            transition={{ duration: 0.18, ease: [0.22, 1, 0.36, 1] }}
   197→            onClick={(e) => e.stopPropagation()}
   198→            className={cn("admin-surface-elevated my-auto w-full overflow-hidden shadow-soft-lg", maxW)}
   199→            role="dialog"
   200→            aria-modal="true"
   201→            aria-labelledby="modal-title"
   202→            tabIndex={-1}
   203→          >
   204→            {/* Header */}
   205→            <div className="flex items-start justify-between gap-4 border-b border-admin-border px-8 py-6">
   206→              <div>
   207→                <h3 id="modal-title" className="font-[family-name:var(--font-playfair)] text-2xl font-semibold text-admin-text">{title}</h3>
   208→                {subtitle && <p className="mt-1 font-sans text-sm text-admin-muted">{subtitle}</p>}
   209→              </div>
   210→              <button
   211→                onClick={onClose}
   212→                className="flex h-9 w-9 shrink-0 items-center justify-center rounded-full border border-admin-border text-admin-muted transition-all hover:border-admin-gold/50 hover:bg-admin-gold/10 hover:text-gold focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-gold"
   213→                aria-label="Close dialog"
   214→              >
   215→                <X className="h-4 w-4" />
   216→              </button>
   217→            </div>
   218→            {/* Body */}
   219→            <div className="max-h-[70vh] overflow-y-auto px-8 py-6">{children}</div>
   220→            {/* Sticky footer */}
   221→            {footer && (
   222→              <div className="sticky bottom-0 flex items-center justify-end gap-2 border-t border-admin-border bg-admin-card/95 px-8 py-4 backdrop-blur-md">
   223→                {footer}
   224→              </div>
   225→            )}
   226→          </motion.div>
   227→        </motion.div>
   228→      )}
   229→    </AnimatePresence>
   230→  );
   231→}

   284→type Option = { value: string; label: string };

   286→export function SearchableSelect({
   287→  label, required, options, value, onChange, placeholder = "Select…", error, searchable = true,
   288→}: {
   289→  label?: string; required?: boolean; options: Option[]; value: string; onChange: (v: string) => void;
   290→  placeholder?: string; error?: string; searchable?: boolean;
   291→}) {
   292→  const [open, setOpen] = useState(false);
   293→  const [query, setQuery] = useState("");
   294→  const [active, setActive] = useState(0);
   295→  const ref = useRef<HTMLDivElement | null>(null);
   296→
   297→  const selected = options.find((o) => o.value === value);
   298→  const filtered = options.filter((o) => o.label.toLowerCase().includes(query.toLowerCase()));
   299→
   300→  useEffect(() => {
   301→    const onDoc = (e: MouseEvent) => { if (ref.current && !ref.current.contains(e.target as Node)) setOpen(false); };
   302→    document.addEventListener("mousedown", onDoc);
   303→    return () => document.removeEventListener("mousedown", onDoc);
   304→  }, []);
   305→
   306→  const choose = (v: string) => { onChange(v); setOpen(false); setQuery(""); };
   307→
   308→  const onKey = (e: React.KeyboardEvent) => {
   309→    if (!open && (e.key === "ArrowDown" || e.key === "Enter")) { setOpen(true); return; }
   310→    if (e.key === "ArrowDown") { e.preventDefault(); setActive((a) => Math.min(a + 1, filtered.length - 1)); }
   311→    if (e.key === "ArrowUp") { e.preventDefault(); setActive((a) => Math.max(a - 1, 0)); }
   312→    if (e.key === "Enter" && open) { e.preventDefault(); if (filtered[active]) choose(filtered[active].value); }
   313→    if (e.key === "Escape") setOpen(false);
   314→  };
   315→
   316→  return (
   317→    <div className="block">
   318→      {label && <span className="mb-1.5 flex items-center gap-1 admin-label">{label}{required && <span className="text-admin-gold">*</span>}</span>}
   319→      <div ref={ref} className="relative" onKeyDown={onKey}>
   320→        <button
   321→          type="button"
   322→          onClick={() => setOpen((o) => !o)}
   323→          className={cn("admin-input flex items-center justify-between text-left", !selected && "text-admin-muted", error && "border-red-500/60")}
   324→        >
   325→          <span className="truncate">{selected ? selected.label : placeholder}</span>
   326→          <ChevronDown className={cn("h-4 w-4 shrink-0 text-admin-muted transition-transform", open && "rotate-180")} />
   327→        </button>
   328→        <AnimatePresence>
   329→          {open && (
   330→            <motion.div
   331→              initial={{ opacity: 0, y: -6, scale: 0.98 }}
   332→              animate={{ opacity: 1, y: 0, scale: 1 }}
   333→              exit={{ opacity: 0, y: -6, scale: 0.98 }}
   334→              transition={{ duration: 0.16, ease: [0.22, 1, 0.36, 1] }}
   335→              className="admin-surface-elevated absolute z-50 mt-2 w-full overflow-hidden p-1.5"
   336→            >
   337→              {searchable && (
   338→                <div className="relative mb-1.5">
   339→                  <Search className="pointer-events-none absolute left-3 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-admin-muted" />
   340→                  <input
   341→                    autoFocus
   342→                    value={query}
   343→                    onChange={(e) => { setQuery(e.target.value); setActive(0); }}
   344→                    placeholder="Search…"
   345→                    className="h-9 w-full rounded-lg border border-admin-border bg-admin-bg/60 pl-9 pr-3 text-sm text-admin-text outline-none focus:border-admin-gold/50"
   346→                  />
   347→                </div>
   348→              )}
   349→              <div className="max-h-56 overflow-y-auto">
   350→                {filtered.length === 0 ? (
   351→                  <p className="px-3 py-6 text-center font-sans text-sm text-admin-muted">No results</p>
   352→                ) : filtered.map((o, i) => (
   353→                  <button
   354→                    key={o.value}
   355→                    type="button"
   356→                    onMouseEnter={() => setActive(i)}
   357→                    onClick={() => choose(o.value)}
   358→                    className={cn(
   359→                      "flex w-full items-center justify-between rounded-lg px-3 py-2 text-left font-sans text-sm transition-colors",
   360→                      i === active ? "bg-admin-gold/15 text-admin-gold" : "text-admin-text hover:bg-white/5"
   361→                    )}
   362→                  >
   363→                    {o.label}
   364→                    {o.value === value && <Check className="h-4 w-4" />}
   365→                  </button>
   366→                ))}
   367→              </div>
   368→            </motion.div>
   369→          )}
   370→        </AnimatePresence>
   371→      </div>
   372→      {error && <p className="mt-1.5 flex items-center gap-1 font-sans text-xs text-red-400"><AlertTriangle className="h-3 w-3" />{error}</p>}
   373→    </div>
   374→  );
   375→}

   407→export function AdminButton({
   408→  children, variant = "solid", className, size = "md", confirm, onConfirm, ...props
   409→}: {
   410→  variant?: "solid" | "outline" | "danger" | "ghost" | "subtle"; size?: "sm" | "md" | "lg";
   411→  confirm?: string; onConfirm?: () => void;
   412→} & React.ButtonHTMLAttributes<HTMLButtonElement>) {
   413→  const sizes = {
   414→    sm: "h-8 px-3 text-[11px]",
   415→    md: "h-10 px-4 text-xs",
   416→    lg: "h-12 px-6 text-sm",
   417→  } as const;
   418→  const styles = {
   419→    solid: "admin-gold-bg text-black font-semibold shadow-gold-glow hover:-translate-y-0.5 hover:shadow-[0_12px_36px_-8px_rgba(212,175,55,0.6)] active:translate-y-0",

   421→    danger: "border border-red-500/30 bg-red-500/10 text-red-400 hover:bg-red-500/20 hover:border-red-500/50",
   422→    ghost: "text-admin-muted hover:bg-white/5 hover:text-admin-text",
   423→    subtle: "bg-white/[0.06] text-admin-text hover:bg-white/10",
   424→  } as const;
   425→
   426→  const handleClick = (e: React.MouseEvent<HTMLButtonElement>) => {
   427→    if (confirm) {
   428→      if (window.confirm(confirm)) { onConfirm?.(); props.onClick?.(e); }
   429→      return;
   430→    }
   431→    props.onClick?.(e);
   432→  };
   433→
   434→  return (
   435→    <button
   436→      {...props}
   437→      onClick={handleClick}
   438→      className={cn(
   439→        "inline-flex items-center justify-center gap-1.5 rounded-xl font-sans uppercase tracking-wider transition-all duration-200 disabled:opacity-50 disabled:pointer-events-none focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-gold focus-visible:ring-offset-2 focus-visible:ring-offset-admin-bg",
   440→        sizes[size], styles[variant], className
   441→      )}
   442→    >
   443→      {children}
   444→    </button>
   445→  );
   446→}

   468→    gold: "border-admin-gold/30 bg-admin-gold/10 text-admin-gold",

   477→    </span>

   479→}

   481→/* =========================================================
   482→   IMAGE UPLOADER — drag & drop, preview, progress, remove/replace
   483→   Uploads to /api/upload (saves to public/uploads/), NEVER Base64 in DB.
   484→   ========================================================= */
   485→export function ImageUploader({
   486→  value, onChange, label = "Image", aspect = "16/10",
   487→}: {
   488→  value: string; onChange: (v: string) => void; label?: string; aspect?: string;
   489→}) {
   490→  const [dragging, setDragging] = useState(false);
   491→  const [progress, setProgress] = useState<number | null>(null);
   492→  const [error, setError] = useState("");
   493→  const [urlMode, setUrlMode] = useState(false);
   494→  const [urlInput, setUrlInput] = useState("");
   495→  const inputRef = useRef<HTMLInputElement | null>(null);
   496→
   497→  const validate = (file: File) => {
   498→    const okType = /image\/(jpeg|jpg|png|webp|gif|avif)/.test(file.type);
   499→    const okSize = file.size <= 6 * 1024 * 1024;
   500→    if (!okType) { setError("Only JPG, PNG, WebP, GIF, AVIF allowed"); return false; }
   501→    if (!okSize) { setError("Max file size is 6MB"); return false; }
   502→    setError(""); return true;
   503→  };
   504→
   505→  const handleFile = useCallback(async (file: File) => {
   506→    if (!validate(file)) return;
   507→    setProgress(0);
   508→    setError("");
   509→    try {
   510→      // Upload to /api/upload — saves to public/uploads/, returns a URL.
   511→      // NEVER store Base64 in the database.
   512→      const url = await apiUpload(file);
   513→      setProgress(100);
   514→      setTimeout(() => setProgress(null), 400);
   515→      onChange(url);
   516→    } catch (e) {
   517→      setProgress(null);
   518→      setError(e instanceof Error ? e.message : "Upload failed");
   519→    }
   520→  }, [onChange]);
   521→
   522→  const onDrop = (e: React.DragEvent) => {
   523→    e.preventDefault(); setDragging(false);
   524→    const file = e.dataTransfer.files?.[0];
   525→    if (file) handleFile(file);
   526→  };
   527→
   528→  const hasImage = !!value;
   529→
   530→  return (
   531→    <div className="block">
   532→      <div className="mb-1.5 flex items-center justify-between">
   533→        <span className="admin-label">{label}</span>
   534→        <button type="button" onClick={() => setUrlMode((m) => !m)} className="font-sans text-[11px] text-admin-gold hover:underline">
   535→          {urlMode ? "Upload file" : "Paste URL"}
   536→        </button>
   537→      </div>
   538→
   539→      {urlMode ? (
   540→        <div className="flex gap-2">
   541→          <input
   542→            value={urlInput}
   543→            onChange={(e) => setUrlInput(e.target.value)}
   544→            placeholder="https://example.com/image.jpg"
   545→            className="admin-input flex-1"
   546→          />
   547→          <AdminButton type="button" variant="subtle" onClick={() => { if (urlInput) { onChange(urlInput); setUrlInput(""); } }}>Set</AdminButton>
   548→        </div>
   549→      ) : !hasImage ? (
   550→        <div
   551→          onDragOver={(e) => { e.preventDefault(); setDragging(true); }}
   552→          onDragLeave={() => setDragging(false)}
   553→          onDrop={onDrop}
   554→          onClick={() => inputRef.current?.click()}
   555→          className={cn(
   556→            "relative flex cursor-pointer flex-col items-center justify-center rounded-xl border-2 border-dashed px-6 py-10 text-center transition-all",
   557→            dragging ? "border-admin-gold bg-admin-gold/10" : "border-admin-border bg-admin-bg/40 hover:border-admin-gold/40 hover:bg-admin-gold/5"
   558→          )}
   559→          style={{ aspectRatio: aspect }}
   560→        >
   561→          <input ref={inputRef} type="file" accept="image/*" className="hidden" onChange={(e) => { const f = e.target.files?.[0]; if (f) handleFile(f); e.target.value = ""; }} />
   562→          <div className="flex h-12 w-12 items-center justify-center rounded-full border border-admin-border bg-admin-gold/10 text-admin-gold">
   563→            <Upload className="h-5 w-5" />
   564→          </div>
   565→          <p className="mt-3 font-sans text-sm font-medium text-admin-text">Drop image here, or <span className="text-admin-gold">browse</span></p>
   566→          <p className="mt-1 font-sans text-xs text-admin-muted">JPG, PNG, WebP, GIF · up to 6MB</p>
   567→        </div>
   568→      ) : (
   569→        <div className="group relative overflow-hidden rounded-xl border border-admin-border" style={{ aspectRatio: aspect }}>
   570→          <img src={value} alt="Preview" className="h-full w-full object-cover" />
   571→          <div className="absolute inset-0 flex items-center justify-center gap-2 bg-black/60 opacity-0 backdrop-blur-sm transition-opacity duration-200 group-hover:opacity-100">
   572→            <AdminButton type="button" size="sm" variant="subtle" onClick={() => inputRef.current?.click()}><ImageIcon className="h-3.5 w-3.5" /> Replace</AdminButton>
   573→            <AdminButton type="button" size="sm" variant="danger" onClick={() => onChange("")}><Trash2 className="h-3.5 w-3.5" /> Remove</AdminButton>
   574→          </div>
   575→          <input ref={inputRef} type="file" accept="image/*" className="hidden" onChange={(e) => { const f = e.target.files?.[0]; if (f) handleFile(f); e.target.value = ""; }} />
   576→        </div>
   577→      )}
   578→
   579→      {/* Progress bar */}
   580→      {progress !== null && (
   581→        <div className="mt-2">
   582→          <div className="h-1.5 w-full overflow-hidden rounded-full bg-white/10">
   583→            <motion.div className="admin-gold-bg h-full" initial={{ width: 0 }} animate={{ width: `${progress}%` }} transition={{ duration: 0.2 }} />
   584→          </div>
   585→          <p className="mt-1 text-center font-sans text-[11px] text-admin-muted">Uploading… {progress}%</p>
   586→        </div>
   587→      )}
   588→      {error && <p className="mt-1.5 flex items-center gap-1 font-sans text-xs text-red-400"><AlertTriangle className="h-3 w-3" />{error}</p>}
   589→    </div>
   590→  );
   591→}

   617→   PAGINATION
   618→   ========================================================= */
   619→export function Pagination({ page, totalPages, onPage }: { page: number; totalPages: number; onPage: (p: number) => void }) {
   620→  if (totalPages <= 1) return null;
   621→  return (
   622→    <div className="flex items-center justify-between gap-3 px-1 py-3">
   623→      <p className="font-sans text-xs text-admin-muted">Page {page} of {totalPages}</p>
   624→      <div className="flex gap-1.5">
   625→        <button onClick={() => onPage(page - 1)} disabled={page <= 1} className="flex h-8 w-8 items-center justify-center rounded-lg border border-admin-border text-admin-muted transition-colors hover:bg-white/5 disabled:opacity-30 disabled:hover:bg-transparent">
   626→          <ChevronLeft className="h-4 w-4" />
   627→        </button>
   628→        {Array.from({ length: totalPages }).map((_, i) => (
   629→          <button
   630→            key={i}
   631→            onClick={() => onPage(i + 1)}
   632→            className={cn("h-8 w-8 rounded-lg border font-sans text-xs transition-colors", page === i + 1 ? "admin-gold-bg border-transparent font-semibold text-black" : "border-admin-border text-admin-muted hover:bg-white/5")}
   633→          >
   634→            {i + 1}
   635→          </button>
   636→        ))}
   637→        <button onClick={() => onPage(page + 1)} disabled={page >= totalPages} className="flex h-8 w-8 items-center justify-center rounded-lg border border-admin-border text-admin-muted transition-colors hover:bg-white/5 disabled:opacity-30 disabled:hover:bg-transparent">
   638→          <ChevronRight className="h-4 w-4" />
   639→        </button>
   640→      </div>
   641→    </div>
   642→  );
   643→}
